package synthesizer

import (
	"bytes"
	"os"
	"path/filepath"
	"testing"
	"time"

	"github.com/router-for-me/CLIProxyAPI/v7/internal/config"
)

// Disabling the removed plugin must ignore both historical Gemini formats
// while keeping Codex usable, without deleting or rewriting credential files.
func TestGatewayLegacyGeminiCredentialsAreIgnoredAndPreserved(t *testing.T) {
	dir := t.TempDir()
	files := map[string][]byte{
		"old-gemini.json": []byte(`{"type":"gemini","access_token":"old","refresh_token":"keep"}`),
		"old-plugin.json": []byte(`{"type":"gemini-cli","access_token":"old","refresh_token":"keep"}`),
		"codex.json":      []byte(`{"type":"codex","access_token":"codex","account_id":"account"}`),
	}
	for name, data := range files {
		if err := os.WriteFile(filepath.Join(dir, name), data, 0600); err != nil {
			t.Fatal(err)
		}
	}
	ctx := &SynthesisContext{AuthDir: dir, Config: &config.Config{}, Now: time.Now()}
	auths, err := NewFileSynthesizer().Synthesize(ctx)
	if err != nil || len(auths) != 1 || auths[0].Provider != "codex" {
		t.Fatalf("expected only Codex, got %v, err %v", auths, err)
	}
	for name, data := range files {
		got, err := os.ReadFile(filepath.Join(dir, name))
		if err != nil || !bytes.Equal(got, data) {
			t.Fatalf("credential %s was changed or removed", name)
		}
		if name != "codex.json" {
			gotAuths, err := SynthesizeAuthFile(ctx, filepath.Join(dir, name), data)
			if err != nil || len(gotAuths) != 0 {
				t.Fatalf("watcher loaded legacy credential %s", name)
			}
		}
	}
}
